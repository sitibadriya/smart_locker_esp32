#include <Arduino.h>
#include <Adafruit_Fingerprint.h>

#define RXD2 16
#define TXD2 17

HardwareSerial mySerial(2);
Adafruit_Fingerprint finger(&mySerial);

uint8_t id;

uint8_t bacaID()
{
    uint8_t p = finger.getImage();

    if (p == FINGERPRINT_OK)
    {
        Serial.println("Gambar sidik jari berhasil diambil.");
        return FINGERPRINT_OK;
    }

    if (p == FINGERPRINT_NOFINGER)
    {
        return FINGERPRINT_NOFINGER;
    }

    Serial.print("Gagal mengambil gambar. Kode: ");
    Serial.println(p);

    return p;
}

uint8_t daftarSidikJari(uint8_t id)
{

    int p = -1;

    Serial.println();
    Serial.println("======================================");
    Serial.print("MENDAFTARKAN FINGERPRINT ID: ");
    Serial.println(id);
    Serial.println("======================================");

    // -----------------------------
    // Scan pertama
    // -----------------------------
    Serial.println("Tempelkan jari Siti Badriya...");

    while (p != FINGERPRINT_OK)
    {
        p = finger.getImage();

        if (p == FINGERPRINT_NOFINGER)
        {
            delay(100);
            continue;
        }

        if (p != FINGERPRINT_OK)
        {
            Serial.print("Gagal membaca jari. Kode: ");
            Serial.println(p);
            return p;
        }
    }

    Serial.println("Jari terdeteksi.");

    p = finger.image2Tz(1);

    if (p != FINGERPRINT_OK)
    {
        Serial.print("Gagal membuat template pertama. Kode: ");
        Serial.println(p);
        return p;
    }

    Serial.println("Template pertama berhasil dibuat.");

    // -----------------------------
    // Lepaskan jari
    // -----------------------------
    Serial.println();
    Serial.println("Angkat jari...");

    delay(1500);

    while (finger.getImage() != FINGERPRINT_NOFINGER)
    {
        delay(100);
    }

    Serial.println("Jari sudah diangkat.");

    // -----------------------------
    // Scan kedua
    // -----------------------------
    Serial.println();
    Serial.println("Tempelkan JARI YANG SAMA lagi...");

    p = -1;

    while (p != FINGERPRINT_OK)
    {
        p = finger.getImage();

        if (p == FINGERPRINT_NOFINGER)
        {
            delay(100);
            continue;
        }

        if (p != FINGERPRINT_OK)
        {
            Serial.print("Gagal membaca jari kedua. Kode: ");
            Serial.println(p);
            return p;
        }
    }

    Serial.println("Jari kedua terdeteksi.");

    p = finger.image2Tz(2);

    if (p != FINGERPRINT_OK)
    {
        Serial.print("Gagal membuat template kedua. Kode: ");
        Serial.println(p);
        return p;
    }

    Serial.println("Template kedua berhasil dibuat.");

    // -----------------------------
    // Cocokkan kedua scan
    // -----------------------------
    Serial.println();
    Serial.println("Mencocokkan kedua scan...");

    p = finger.createModel();

    if (p == FINGERPRINT_OK)
    {
        Serial.println("Sidik jari berhasil dicocokkan.");
    }
    else if (p == FINGERPRINT_ENROLLMISMATCH)
    {
        Serial.println("KEDUA SCAN TIDAK COCOK!");
        return p;
    }
    else
    {
        Serial.print("Gagal membuat model. Kode: ");
        Serial.println(p);
        return p;
    }

    // -----------------------------
    // Simpan ke ID
    // -----------------------------
    p = finger.storeModel(id);

    if (p == FINGERPRINT_OK)
    {
        Serial.println();
        Serial.println("**************************************");
        Serial.println("SIDIK JARI BERHASIL DISIMPAN!");
        Serial.print("Fingerprint ID : ");
        Serial.println(id);
        Serial.println("Pemilik        : Siti Badriya");
        Serial.println("Locker         : L01");
        Serial.println("**************************************");

        return FINGERPRINT_OK;
    }

    Serial.print("Gagal menyimpan fingerprint. Kode: ");
    Serial.println(p);

    return p;
}

void setup()
{

    Serial.begin(115200);
    delay(1000);

    Serial.println();
    Serial.println("======================================");
    Serial.println(" SMART LOCKER");
    Serial.println(" ENROLL 5 JARI SITI BADRIYA");
    Serial.println("======================================");

    mySerial.begin(57600, SERIAL_8N1, RXD2, TXD2);

    finger.begin(57600);

    delay(1000);

    Serial.println();
    Serial.println("Memeriksa sensor fingerprint...");

    if (finger.verifyPassword())
    {
        Serial.println("Sensor fingerprint TERDETEKSI!");
    }
    else
    {
        Serial.println("Sensor fingerprint TIDAK TERDETEKSI!");
        Serial.println("Periksa wiring TX/RX/GND.");

        while (true)
        {
            delay(1000);
        }
    }

    Serial.print("Kapasitas sensor: ");
    Serial.println(finger.capacity);

    Serial.println();
    Serial.println("======================================");
    Serial.println(" SIAP MENDAFTARKAN 5 JARI");
    Serial.println("======================================");

    Serial.println();
    Serial.println("ID 1 = Jari pertama");
    Serial.println("ID 2 = Jari kedua");
    Serial.println("ID 3 = Jari ketiga");
    Serial.println("ID 4 = Jari keempat");
    Serial.println("ID 5 = Jari kelima");

    delay(2000);
}

void loop()
{

    static uint8_t idBerikutnya = 1;

    if (idBerikutnya > 5)
    {

        Serial.println();
        Serial.println("======================================");
        Serial.println(" SEMUA 5 JARI BERHASIL DIDAFTARKAN");
        Serial.println("======================================");

        Serial.println();
        Serial.println("Pemilik : Siti Badriya");
        Serial.println("Locker  : L01");

        Serial.println();
        Serial.println("ID 1 -> Siti Badriya");
        Serial.println("ID 2 -> Siti Badriya");
        Serial.println("ID 3 -> Siti Badriya");
        Serial.println("ID 4 -> Siti Badriya");
        Serial.println("ID 5 -> Siti Badriya");

        Serial.println();
        Serial.println("Enrollment selesai.");
        Serial.println("Jangan mendaftarkan ulang ID 1-5.");

        while (true)
        {
            delay(1000);
        }
    }

    uint8_t hasil = daftarSidikJari(idBerikutnya);

    if (hasil == FINGERPRINT_OK)
    {
        idBerikutnya++;

        Serial.println();
        Serial.println("Lanjut ke ID berikutnya...");
        delay(2000);
    }
    else
    {
        Serial.println();
        Serial.println("Enrollment gagal.");
        Serial.println("Silakan ulangi ID yang sama.");

        delay(2000);
    }
}
