#include <Arduino.h>
#include <Adafruit_Fingerprint.h>

#define RXD2 16
#define TXD2 17

HardwareSerial mySerial(2);
Adafruit_Fingerprint finger = Adafruit_Fingerprint(&mySerial);

void setup()
{
    Serial.begin(115200);
    delay(1000);

    Serial.println();
    Serial.println("=== TEST PENGENALAN FINGERPRINT ===");

    mySerial.begin(57600, SERIAL_8N1, RXD2, TXD2);
    finger.begin(57600);

    if (finger.verifyPassword())
    {
        Serial.println("Sensor Fingerprint terdeteksi!");
        Serial.println("Silakan tempelkan jari yang sudah didaftarkan.");
    }
    else
    {
        Serial.println("Sensor Fingerprint tidak terdeteksi!");

        while (1)
        {
            delay(1000);
        }
    }
}

void loop()
{
    uint8_t p = finger.getImage();

    if (p == FINGERPRINT_NOFINGER)
    {
        delay(100);
        return;
    }

    if (p != FINGERPRINT_OK)
    {
        Serial.println("Gagal membaca sidik jari.");
        delay(500);
        return;
    }

    Serial.println();
    Serial.println("Sidik jari terdeteksi!");

    p = finger.image2Tz();

    if (p != FINGERPRINT_OK)
    {
        Serial.println("Gagal memproses sidik jari.");
        delay(1000);
        return;
    }

    Serial.println("Mencari sidik jari...");

    p = finger.fingerSearch();

    if (p == FINGERPRINT_OK)
    {
        Serial.println(">>> SIDIK JARI DITEMUKAN! <<<");

        Serial.print("Fingerprint ID : ");
        Serial.println(finger.fingerID);

        Serial.print("Confidence      : ");
        Serial.println(finger.confidence);

        if (finger.fingerID == 1)
        {
            Serial.println("STATUS: PEMILIK LOKER L01");
            Serial.println("AKSES: GRANTED");
        }
        else
        {
            Serial.println("STATUS: BUKAN PEMILIK LOKER L01");
            Serial.println("AKSES: DENIED");
        }
    }
    else if (p == FINGERPRINT_NOTFOUND)
    {
        Serial.println(">>> SIDIK JARI TIDAK DITEMUKAN <<<");
        Serial.println("AKSES: DENIED");
    }
    else
    {
        Serial.println("Gagal melakukan pencarian sidik jari.");
    }

    Serial.println();
    Serial.println("Silakan angkat jari...");

    delay(2000);
}
