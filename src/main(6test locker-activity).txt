#define USER_EMAIL "smartlocker.esp32@gmail.com"

#define FIREBASE_PROJECT_ID "smart-locker-iot-10a0e"

// ======================================

FirebaseApp app;

AsyncClient aClient(ssl_client);

Firestore::Documents Docs;

// ======================================
// STATUS
// ======================================

bool dataSent = false;

// ======================================
// CALLBACK AUTH
// ======================================

void authCallback(AsyncResult &aResult)
{
    if (aResult.isEvent())
    {
        Serial.printf(
            "Event task: %s, msg: %s, code: %d\n",
            aResult.uid().c_str(),
            aResult.appEvent().message().c_str(),
            aResult.appEvent().code());
    }

    if (aResult.isDebug())
    {
        Serial.printf(
            "Debug task: %s, msg: %s\n",
            aResult.uid().c_str(),
            aResult.debug().c_str());
    }

    if (aResult.isError())
    {
        Serial.printf(
            "Error task: %s, msg: %s, code: %d\n",
            aResult.uid().c_str(),
            aResult.error().message().c_str(),
            aResult.error().code());
    }
}

// ======================================
// SETUP
// ======================================

void setup()
{
    Serial.begin(115200);

    delay(1000);

    Serial.println();
    Serial.println("======================================");
    Serial.println(" SMART LOCKER - FIRESTORE TEST");
    Serial.println("======================================");

    // ==================================
    // WIFI
    // ==================================

    WiFi.begin(WIFI_SSID, WIFI_PASSWORD);

    Serial.print("Menghubungkan ke Wi-Fi");

    while (WiFi.status() != WL_CONNECTED)
    {
        Serial.print(".");
        delay(300);
    }

    Serial.println();
    Serial.println("Wi-Fi TERHUBUNG!");

    Serial.print("IP ESP32: ");
    Serial.println(WiFi.localIP());

    Serial.println();

    // ==================================
    // SSL
    // ==================================

    ssl_client.setInsecure();

    // ==================================
    // FIREBASE AUTH
    // ==================================

    Serial.println("Menginisialisasi Firebase...");

    initializeApp(
        aClient,
        app,
        getAuth(user_auth),
        authCallback,
        "authTask");

    Serial.println("Firebase initialization dimulai...");

    // ==================================
    // HUBUNGKAN FIRESTORE KE APP
    // ==================================

    app.getApp<Firestore::Documents>(Docs);

    Serial.println("Firestore siap digunakan.");
}

// ======================================
// LOOP
// ======================================

void loop()
{
    // Menjalankan Firebase
    app.loop();

    // Menjalankan Firestore
    Docs.loop();

    // ==================================
    // TUNGGU AUTH BERHASIL
    // ==================================

    if (app.ready() && !dataSent)
    {
        dataSent = true;

        Serial.println();
        Serial.println("======================================");
        Serial.println(" FIREBASE AUTHENTICATION BERHASIL");
        Serial.println("======================================");

        Serial.print("User UID: ");
        Serial.println(app.getUid());

        Serial.println();
        Serial.println("Mengirim data test ke Firestore...");

        // ==================================
        // DATA TEST
        // ==================================

        Values::StringValue lockerId("L01");

        Values::StringValue deviceId(
            "esp32-locker-01");

        Values::StringValue fingerprintId(
            "FP01");

        Values::StringValue status(
            "granted");

        Values::StringValue accessMethod(
            "fingerprint");

        Values::StringValue dataFp(
            "jari pemilik");

        // ==================================
        // BUAT DOCUMENT
        // ==================================

        Document<Values::Value> doc(
            "lockerId",
            Values::Value(lockerId));

        doc.add(
            "deviceId",
            Values::Value(deviceId));

        doc.add(
            "fingerprintId",
            Values::Value(fingerprintId));

        doc.add(
            "status",
            Values::Value(status));

        doc.add(
            "accessMethod",
            Values::Value(accessMethod));

        doc.add(
            "Data_FP",
            Values::Value(dataFp));

        // ==================================
        // PATH DOCUMENT
        // ==================================

        String documentPath =
            "locker-activity";

        // ==================================
        // KIRIM KE FIRESTORE
        // ==================================

        Serial.println("Membuat document...");

        String payload = Docs.createDocument(
            aClient,
            Firestore::Parent(FIREBASE_PROJECT_ID),
            documentPath,
            DocumentMask(),
            doc);

        // ==================================
        // CEK HASIL
        // ==================================

        if (aClient.lastError().code() == 0)
        {
            Serial.println();
            Serial.println("======================================");
            Serial.println(" DATA BERHASIL DIKIRIM!");
            Serial.println("======================================");

            Serial.println(payload);
        }
        else
        {
            Serial.println();
            Serial.println("======================================");
            Serial.println(" GAGAL MENGIRIM DATA");
            Serial.println("======================================");

            Serial.print("Error code: ");
            Serial.println(
                aClient.lastError().code());

            Serial.print("Error message: ");
            Serial.println(
                aClient.lastError().message());
        }
    }

    delay(10);
}
