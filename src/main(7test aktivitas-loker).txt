// ======================================
// CALLBACK AUTH
// ======================================

void authCallback(AsyncResult &aResult)
{
    if (aResult.isEvent())
    {
        Serial.printf(
            "Event task: %s, msg: %s, code: %d\n",
            aResult.uid().c_str(),
            aResult.appEvent().message().c_str(),
            aResult.appEvent().code());
    }

    if (aResult.isDebug())
    {
        Serial.printf(
            "Debug task: %s, msg: %s\n",
            aResult.uid().c_str(),
            aResult.debug().c_str());
    }

    if (aResult.isError())
    {
        Serial.printf(
            "Error task: %s, msg: %s, code: %d\n",
            aResult.uid().c_str(),
            aResult.error().message().c_str(),
            aResult.error().code());
    }
}

// ======================================
// MENDAPATKAN WAKTU UTC
// ======================================

String dapatkanWaktuUTC()
{
    time_t sekarang = time(nullptr);

    struct tm waktu;

    gmtime_r(&sekarang, &waktu);

    char buffer[30];

    strftime(
        buffer,
        sizeof(buffer),
        "%Y-%m-%dT%H:%M:%SZ",
        &waktu);

    return String(buffer);
}
